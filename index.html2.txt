                                data-id="${product.id}" 
                                data-name="${product.name} (${category})" 
                                data-price="${product.price}" 
                                data-image="${product.image}"
                                ${product.game ? `data-game="${product.game}"` : ''}>
                            Добавить в корзину
                        </button>
                    `;
                    productsContainer.appendChild(productCard);
                });
            }
            
            // Добавляем обработчики для кнопок "Добавить в корзину"
            document.querySelectorAll(".add-to-cart").forEach(button => {
                button.addEventListener("click", addToCartHandler);
            });
        }

        // Обработчик добавления в корзину
        function addToCartHandler() {
            const id = this.getAttribute("data-id");
            const name = this.getAttribute("data-name");
            const price = parseInt(this.getAttribute("data-price"));
            const image = this.getAttribute("data-image");
            const game = this.getAttribute("data-game");
            
            // Проверяем, есть ли товар уже в корзине
            const existingItem = cart.find(item => item.id === id);
            
            if (existingItem) {
                existingItem.quantity += 1;
            } else {
                cart.push({
                    id,
                    name,
                    price,
                    image,
                    game,
                    quantity: 1
                });
            }
            
            // Анимация добавления
            this.classList.add("added-to-cart");
            setTimeout(() => {
                this.classList.remove("added-to-cart");
            }, 500);
            
            updateCartCount();
            renderCart();
            checkGameSpecificFields();
        }

        // Проверка на необходимость показать дополнительные поля
        function checkGameSpecificFields() {
            const hasBrawlItems = cart.some(item => item.game === 'brawl');
            const hasTgItems = cart.some(item => item.game === 'telegram');
            
            brawlFields.style.display = hasBrawlItems ? 'block' : 'none';
            tgFields.style.display = hasTgItems ? 'block' : 'none';
            
            if (hasBrawlItems) brawlTagInput.required = true;
            if (hasTgItems) tgUsernameInput.required = true;
        }

        // Открытие/закрытие корзины
        document.querySelector(".cart-btn").addEventListener("click", () => {
            cartModal.style.display = "block";
            renderCart();
            checkGameSpecificFields();
        });
        
        document.querySelector(".close-cart").addEventListener("click", () => {
            cartModal.style.display = "none";
        });

        // Кнопка показа каталога
        showCatalogBtn.addEventListener("click", initCatalog);

        // Применение промокода
        applyPromoBtn.addEventListener("click", () => {
            const promo = promoCodeInput.value.trim().toUpperCase();
            
            if (PROMO_CODES[promo]) {
                discount = PROMO_CODES[promo];
                appliedPromo = promo;
                promoMessage.textContent = `Промокод "${promo}" применен! Скидка ${discount}%`;
                promoMessage.style.color = "var(--discount-color)";
                renderCart();
            } else {
                promoMessage.textContent = "Неверный промокод";
                promoMessage.style.color = "#ff4444";
            }
        });

        // Обновление счетчика корзины
        function updateCartCount() {
            const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
            cartCountElement.textContent = totalItems;
        }
        
        // Отрисовка корзины
        function renderCart() {
            if (cart.length === 0) {
                cartItemsContainer.innerHTML = "<p>Корзина пуста</p>";
                cartTotalElement.textContent = "0";
                checkoutSection.style.display = "none";
                discountApplied.textContent = "";
                return;
            }
            
            checkoutSection.style.display = "block";
            successMessage.style.display = "none";
            
            cartItemsContainer.innerHTML = "";
            
            let subtotal = 0;
            
            cart.forEach(item => {
                const itemElement = document.createElement("div");
                itemElement.className = "cart-item";
                itemElement.innerHTML = `
                    <div class="cart-item-info">
                        <img src="${item.image}" alt="${item.name}" class="cart-item-img">
                        <div>
                            <div>${item.name}</div>
                            <div>${item.price} ₽</div>
                        </div>
                    </div>
                    <div class="quantity-control">
                        <button class="quantity-btn minus" data-id="${item.id}">-</button>
                        <span>${item.quantity}</span>
                        <button class="quantity-btn plus" data-id="${item.id}">+</button>
                        <span class="remove-item" data-id="${item.id}">✕</span>
                    </div>
                `;
                
                cartItemsContainer.appendChild(itemElement);
                subtotal += item.price * item.quantity;
            });
            
            // Применение скидки
            let total = subtotal;
            if (discount > 0) {
                const discountAmount = Math.round(subtotal * discount / 100);
                total = subtotal - discountAmount;
                discountApplied.textContent = `Скидка ${discount}%: -${discountAmount} ₽ (Промокод: ${appliedPromo})`;
            } else {
                discountApplied.textContent = "";
            }
            
            cartTotalElement.textContent = total;
            
            // Обработчики для кнопок +/-
            document.querySelectorAll(".quantity-btn.minus").forEach(btn => {
                btn.addEventListener("click", function() {
                    const id = this.getAttribute("data-id");
                    const item = cart.find(item => item.id === id);
                    
                    if (item.quantity > 1) {
                        item.quantity -= 1;
                    } else {
                        cart = cart.filter(item => item.id !== id);
                    }
                    
                    updateCartCount();
                    renderCart();
                    checkGameSpecificFields();
                });
            });
            
            document.querySelectorAll(".quantity-btn.plus").forEach(btn => {
                btn.addEventListener("click", function() {
                    const id = this.getAttribute("data-id");
                    const item = cart.find(item => item.id === id);
                    item.quantity += 1;
                    
                    updateCartCount();
                    renderCart();
                });
            });
            
            document.querySelectorAll(".remove-item").forEach(btn => {
                btn.addEventListener("click", function() {
                    const id = this.getAttribute("data-id");
                    cart = cart.filter(item => item.id !== id);
                    
                    updateCartCount();
                    renderCart();
                    checkGameSpecificFields();
                });
            });
        }
        
        // Оформление заказа
        submitOrderBtn.addEventListener("click", function() {
            const playerId = playerIdInput.value.trim();
            const brawlTag = brawlTagInput.value.trim();
            const tgUsername = tgUsernameInput.value.trim();
            
            if (!playerId) {
                alert("Пожалуйста, введите ваш игровой ID или никнейм!");
                return;
            }
            
            if (cart.length === 0) {
                alert("Корзина пуста!");
                return;
            }
            
            // Проверяем дополнительные поля
            const hasBrawlItems = cart.some(item => item.game === 'brawl');
            const hasTgItems = cart.some(item => item.game === 'telegram');
            
            if (hasBrawlItems && !brawlTag) {
                alert("Пожалуйста, введите ваш тег Brawl Stars!");
                return;
            }
            
            if (hasTgItems && !tgUsername) {
                alert("Пожалуйста, введите ваш Telegram username!");
                return;
            }
            
            // Формируем сообщение для Telegram
            let message = `🛒 Новый заказ в Game Store\n\n`;
            
            cart.forEach(item => {
                message += `🎮 ${item.name} - ${item.quantity} шт. × ${item.price} ₽ = ${item.quantity * item.price} ₽\n`;
            });
            
            const subtotal = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);
            let total = subtotal;
            
            if (discount > 0) {
                const discountAmount = Math.round(subtotal * discount / 100);
                total = subtotal - discountAmount;
                message += `\n🎫 Промокод: ${appliedPromo} (скидка ${discount}%)\n`;
                message += `➖ Скидка: ${discountAmount} ₽\n`;
            }
            
            message += `\n💰 Итого: ${total} ₽\n`;
            message += `🆔 ID игрока: ${playerId}\n`;
            
            if (hasBrawlItems) {
                message += `🎮 Brawl Stars ID: ${brawlTag}\n`;
            }
            
            if (hasTgItems) {
                message += `📱 Telegram: ${tgUsername}\n`;
            }
            
            message += `⏰ Время: ${new Date().toLocaleString()}\n\n`;
            message += `💳 Оплатить на:\nQIWI: 79179303055\nТинькофф: 2200701930415515`;
            
            // Отправляем в Telegram
            fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage?chat_id=${CHAT_ID}&text=${encodeURIComponent(message)}`)
                .then(response => {
                    if (response.ok) {
                        // Показываем сообщение об успехе
                        checkoutSection.style.display = "none";
                        successMessage.style.display = "block";
                        
                        // Очищаем корзину
                        cart = [];
                        discount = 0;
                        appliedPromo = '';
                        promoCodeInput.value = '';
                        promoMessage.textContent = '';
                        playerIdInput.value = '';
                        brawlTagInput.value = '';
                        tgUsernameInput.value = '';
                        updateCartCount();
                    } else {
                        alert("Ошибка при отправке заказа. Пожалуйста, попробуйте еще раз.");
                    }
                })
                .catch(error => {
                    console.error("Error:", error);
                    alert("Произошла ошибка. Пожалуйста, свяжитесь с нами через Telegram.");
                });
        });

        // Инициализация при загрузке страницы
        document.addEventListener('DOMContentLoaded', initCatalog);
    </script>
